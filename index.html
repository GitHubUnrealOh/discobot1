const { Client, GatewayIntentBits } = require('discord.js'); const client = new Client({ intents: [ GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages, GatewayIntentBits.MessageContent ] }); let activeEvent = null; // SURVIVAL SYSTEM const survival = {}; const SURVIVE_ROLES = [ "I SURVIVED 1", "I SURVIVED 2", "I SURVIVED 3" ]; // EVENTS (GENERIC) const events = [ { type: "word", text: "⚡ FIRST to type the secret word wins something...", word: "typeit", rewardRole: "Reward 1" }, { type: "firstAny", text: "💀 FIRST person to send a message triggers something...", punishRole: "Punish 1", timeout: 10 * 1000 }, { type: "reaction", text: "🎯 FIRST to react with 👁️ wins something...", emoji: "👁️", rewardRole: "Reward 1" }, { type: "vote", text: "🗳️ Nominate someone. Most mentioned gets affected...", duration: 15000, punishRole: "Punish 1" } ]; // SURVIVAL LEVEL UP async function upgradeSurvivor(member, message) { const id = member.id; if (!survival[id]) survival[id] = 0; if (survival[id] >= 3) { message.channel.send(`🏆 ${member} is already MAX SURVIVOR.`); return; } if (survival[id] > 0) { const oldRole = message.guild.roles.cache.find( r => r.name === SURVIVE_ROLES[survival[id] - 1] ); if (oldRole) await member.roles.remove(oldRole); } survival[id]++; const newRole = message.guild.roles.cache.find( r => r.name === SURVIVE_ROLES[survival[id] - 1] ); if (newRole) await member.roles.add(newRole); message.channel.send(`🛡️ ${member} survived → **${SURVIVE_ROLES[survival[id] - 1]}**`); } // MAIN TRIGGER client.on('messageCreate', async (message) => { if (message.author.bot) return; // START EVENT if (message.mentions.has(client.user) && !activeEvent) { const event = events[Math.floor(Math.random() * events.length)]; activeEvent = event; message.channel.send(event.text); // ====================== // WORD EVENT // ====================== if (event.type === "word") { const collector = message.channel.createMessageCollector({ time: 20000 }); collector.on('collect', async (msg) => { if (msg.content.toLowerCase().includes(event.word)) { collector.stop(); const member = await message.guild.members.fetch(msg.author.id); const role = message.guild.roles.cache.find(r => r.name === event.rewardRole); if (role) await member.roles.add(role); message.channel.send(`🎉 ${member} was FIRST and triggered the event!`); await upgradeSurvivor(member, message); activeEvent = null; } }); } // ====================== // FIRST MESSAGE EVENT // ====================== if (event.type === "firstAny") { const collector = message.channel.createMessageCollector({ time: 20000 }); collector.on('collect', async (msg) => { collector.stop(); const member = await message.guild.members.fetch(msg.author.id); const role = message.guild.roles.cache.find(r => r.name === event.punishRole); if (role) await member.roles.add(role); await member.timeout(event.timeout, "Lost event"); message.channel.send(`💀 ${member} was FIRST and triggered a punishment.`); await upgradeSurvivor(member, message); activeEvent = null; }); } // ====================== // REACTION EVENT // ====================== if (event.type === "reaction") { const sent = await message.channel.send("React fast!"); await sent.react(event.emoji); const filter = (reaction, user) => reaction.emoji.name === event.emoji && !user.bot; const collector = sent.createReactionCollector({ filter, max: 1, time: 20000 }); collector.on('collect', async (reaction, user) => { const member = await message.guild.members.fetch(user.id); const role = message.guild.roles.cache.find(r => r.name === event.rewardRole); if (role) await member.roles.add(role); message.channel.send(`🎯 ${member} reacted FIRST and won the event!`); await upgradeSurvivor(member, message); activeEvent = null; }); } // ====================== // VOTING EVENT // ====================== if (event.type === "vote") { const votes = {}; const collector = message.channel.createMessageCollector({ time: event.duration }); collector.on('collect', msg => { msg.mentions.users.forEach(u => { votes[u.id] = (votes[u.id] || 0) + 1; }); }); collector.on('end', async () => { let max = 0; let loserId = null; for (const id in votes) { if (votes[id] > max) { max = votes[id]; loserId = id; } } if (!loserId) { message.channel.send("🤷 No result."); activeEvent = null; return; } const member = await message.guild.members.fetch(loserId); const role = message.guild.roles.cache.find(r => r.name === event.punishRole); if (role) await member.roles.add(role); await member.timeout(30 * 60 * 1000, "Vote event"); message.channel.send(`💀 ${member} was voted out.`); await upgradeSurvivor(member, message); activeEvent = null; }); } } }); client.login("YOUR_BOT_TOKEN");
